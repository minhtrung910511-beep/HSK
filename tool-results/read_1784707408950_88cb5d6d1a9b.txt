   227→const STORAGE_KEY = "hsk1-progress-v1";
   228→const DAY_MS = 24 * 60 * 60 * 1000;
   229→
   230→function todayKey() {
   231→  const d = new Date();
   232→  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;
   233→}

   235→function defaultProgress() {
   236→  return {
   237→    cards: {}, learnedWordIds: [],
   238→    streak: { current: 0, longest: 0, lastStudyDate: "" },
   239→    totalReviews: 0, correctReviews: 0, dailyStats: {},
   240→    quizHighScore: 0, matchingHighScore: 0,
   241→  };
   242→}
   243→
   244→function loadProgress() {
   245→  try {
   246→    const raw = localStorage.getItem(STORAGE_KEY);
   247→    if (!raw) return defaultProgress();
   248→    return { ...defaultProgress(), ...JSON.parse(raw) };
   249→  } catch (e) { return defaultProgress(); }
   250→}
   251→
   252→function saveProgress(p) {
   253→  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(p)); } catch(e) {}
   254→}

   256→function newCard(wordId) {
   257→  return { wordId, repetitions: 0, interval: 0, easeFactor: 2.5, dueDate: Date.now(), lastReviewed: null, history: [] };
   258→}
   259→
   260→function reviewCard(card, grade, now) {
   261→  now = now || Date.now();
   262→  let repetitions = card.repetitions;
   263→  let interval = card.interval;
   264→  let easeFactor = card.easeFactor;
   265→  const newEase = Math.max(1.3, easeFactor + (0.1 - (5 - grade) * (0.08 + (5 - grade) * 0.02)));
   266→  if (grade < 3) {
   267→    repetitions = 0; interval = 1;
   268→  } else {
   269→    repetitions += 1;
   270→    if (repetitions === 1) interval = 1;
   271→    else if (repetitions === 2) interval = 3;
   272→    else interval = Math.round(interval * newEase);
   273→  }
   274→  return Object.assign({}, card, {
   275→    repetitions, interval, easeFactor: newEase,
   276→    dueDate: now + interval * DAY_MS,
   277→    lastReviewed: now,
   278→    history: card.history.concat([{ date: now, grade }]),
   279→  });
   280→}

   282→function getDueCards(progress, allWordIds) {
   283→  const now = Date.now();
   284→  return allWordIds.filter(function(wordId) {
   285→    const card = progress.cards[wordId];
   286→    return !card || card.dueDate <= now;
   287→  });
   288→}

   290→function gradeWord(progress, wordId, grade) {
   291→  const now = Date.now();
   292→  const existing = progress.cards[wordId] || newCard(wordId);
   293→  const updated = reviewCard(existing, grade, now);
   294→  const today = todayKey();
   295→  const dailyStats = Object.assign({}, progress.dailyStats);
   296→  if (!dailyStats[today]) dailyStats[today] = { studied: 0, correct: 0 };
   297→  dailyStats[today].studied += 1;
   298→  if (grade >= 3) dailyStats[today].correct += 1;
   299→  const lastDate = progress.streak.lastStudyDate;
   300→  let current = progress.streak.current;
   301→  let longest = progress.streak.longest;
   302→  if (lastDate !== today) {
   303→    const y = new Date(now - DAY_MS);
   304→    const yKey = `${y.getFullYear()}-${String(y.getMonth()+1).padStart(2,"0")}-${String(y.getDate()).padStart(2,"0")}`;
   305→    current = lastDate === yKey ? current + 1 : 1;
   306→    if (current > longest) longest = current;
   307→  }
   308→  const learnedSet = new Set(progress.learnedWordIds);
   309→  if (grade >= 3) learnedSet.add(wordId);
   310→  return Object.assign({}, progress, {
   311→    cards: Object.assign({}, progress.cards, { [wordId]: updated }),
   312→    learnedWordIds: Array.from(learnedSet),
   313→    totalReviews: progress.totalReviews + 1,
   314→    correctReviews: progress.correctReviews + (grade >= 3 ? 1 : 0),
   315→    streak: { current, longest, lastStudyDate: today },
   316→    dailyStats,
   317→  });
   318→}

   320→function markLearned(progress, wordId) {
   321→  const s = new Set(progress.learnedWordIds);
   322→  s.add(wordId);
   323→  return Object.assign({}, progress, { learnedWordIds: Array.from(s) });
   324→}

   326→// ==================== Hook: useProgress ====================
   327→function useProgress() {
   328→  const [progress, setProgress] = useState(null);
   329→  const [hydrated, setHydrated] = useState(false);
   330→  useEffect(function() {
   331→    setProgress(loadProgress());
   332→    setHydrated(true);
   333→  }, []);
   334→  const grade = useCallback(function(wordId, g) {
   335→    setProgress(function(p) { if (!p) return p; const next = gradeWord(p, wordId, g); saveProgress(next); return next; });
   336→  }, []);
   337→  const markLearnedWord = useCallback(function(wordId) {
   338→    setProgress(function(p) { if (!p) return p; const next = markLearned(p, wordId); saveProgress(next); return next; });
   339→  }, []);
   340→  const recordQuizScore = useCallback(function(score) {
   341→    setProgress(function(p) { if (!p) return p; const next = score > p.quizHighScore ? Object.assign({}, p, { quizHighScore: score }) : p; saveProgress(next); return next; });
   342→  }, []);
   343→  const recordMatchingScore = useCallback(function(score) {
   344→    setProgress(function(p) { if (!p) return p; const next = score > p.matchingHighScore ? Object.assign({}, p, { matchingHighScore: score }) : p; saveProgress(next); return next; });
   345→  }, []);
   346→  const reset = useCallback(function() {
   347→    const fresh = defaultProgress(); saveProgress(fresh); setProgress(fresh);
   348→  }, []);
   349→  return { progress, hydrated, grade, markLearnedWord, recordQuizScore, recordMatchingScore, reset };
   350→}
