   227→const STORAGE_KEY = "hsk1-progress-v1";
   228→const DAY_MS = 24 * 60 * 60 * 1000;
   229→
   230→function todayKey() {
   231→  const d = new Date();
   232→  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;
   233→}

   235→function defaultProgress() {
   236→  return {
   237→    cards: {}, learnedWordIds: [],
   238→    streak: { current: 0, longest: 0, lastStudyDate: "" },
   239→    totalReviews: 0, correctReviews: 0, dailyStats: {},
   240→    quizHighScore: 0, matchingHighScore: 0,
   241→  };
   242→}
   243→
   244→function loadProgress() {
   245→  try {
   246→    const raw = localStorage.getItem(STORAGE_KEY);
   247→    if (!raw) return defaultProgress();
   248→    return { ...defaultProgress(), ...JSON.parse(raw) };
   249→  } catch (e) { return defaultProgress(); }
   250→}
   251→
   252→function saveProgress(p) {
   253→  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(p)); } catch(e) {}
   254→}

   256→function newCard(wordId) {
   257→  return { wordId, repetitions: 0, interval: 0, easeFactor: 2.5, dueDate: Date.now(), lastReviewed: null, history: [] };
   258→}
   259→
   260→function reviewCard(card, grade, now) {
   261→  now = now || Date.now();
   262→  let repetitions = card.repetitions;
   263→  let interval = card.interval;
   264→  let easeFactor = card.easeFactor;
   265→  const newEase = Math.max(1.3, easeFactor + (0.1 - (5 - grade) * (0.08 + (5 - grade) * 0.02)));
   266→  if (grade < 3) {
   267→    repetitions = 0; interval = 1;
   268→  } else {
   269→    repetitions += 1;
   270→    if (repetitions === 1) interval = 1;
   271→    else if (repetitions === 2) interval = 3;
   272→    else interval = Math.round(interval * newEase);
   273→  }
   274→  return Object.assign({}, card, {
   275→    repetitions, interval, easeFactor: newEase,
   276→    dueDate: now + interval * DAY_MS,
   277→    lastReviewed: now,
   278→    history: card.history.concat([{ date: now, grade }]),
   279→  });
   280→}
   281→
   282→function getDueCards(progress, allWordIds) {
   283→  const now = Date.now();
   284→  return allWordIds.filter(function(wordId) {
   285→    const card = progress.cards[wordId];
   286→    return !card || card.dueDate <= now;
   287→  });
   288→}
   289→
   290→function gradeWord(progress, wordId, grade) {
   291→  const now = Date.now();
   292→  const existing = progress.cards[wordId] || newCard(wordId);
   293→  const updated = reviewCard(existing, grade, now);
   294→  const today = todayKey();
   295→  const dailyStats = Object.assign({}, progress.dailyStats);
   296→  if (!dailyStats[today]) dailyStats[today] = { studied: 0, correct: 0 };
   297→  dailyStats[today].studied += 1;
   298→  if (grade >= 3) dailyStats[today].correct += 1;
   299→  const lastDate = progress.streak.lastStudyDate;
   300→  let current = progress.streak.current;
   301→  let longest = progress.streak.longest;
   302→  if (lastDate !== today) {
   303→    const y = new Date(now - DAY_MS);
   304→    const yKey = `${y.getFullYear()}-${String(y.getMonth()+1).padStart(2,"0")}-${String(y.getDate()).padStart(2,"0")}`;
   305→    current = lastDate === yKey ? current + 1 : 1;
   306→    if (current > longest) longest = current;
   307→  }
   308→  const learnedSet = new Set(progress.learnedWordIds);
   309→  if (grade >= 3) learnedSet.add(wordId);
   310→  return Object.assign({}, progress, {
   311→    cards: Object.assign({}, progress.cards, { [wordId]: updated }),
   312→    learnedWordIds: Array.from(learnedSet),
   313→    totalReviews: progress.totalReviews + 1,
   314→    correctReviews: progress.correctReviews + (grade >= 3 ? 1 : 0),
   315→    streak: { current, longest, lastStudyDate: today },
   316→    dailyStats,
   317→  });
   318→}

   320→function markLearned(progress, wordId) {
   321→  const s = new Set(progress.learnedWordIds);
   322→  s.add(wordId);
   323→  return Object.assign({}, progress, { learnedWordIds: Array.from(s) });
   324→}

   326→// ==================== Hook: useProgress ====================
   327→function useProgress() {
   328→  const [progress, setProgress] = useState(null);
   329→  const [hydrated, setHydrated] = useState(false);
   330→  useEffect(function() {
   331→    setProgress(loadProgress());
   332→    setHydrated(true);
   333→  }, []);
   334→  const grade = useCallback(function(wordId, g) {
   335→    setProgress(function(p) { if (!p) return p; const next = gradeWord(p, wordId, g); saveProgress(next); return next; });
   336→  }, []);
   337→  const markLearnedWord = useCallback(function(wordId) {
   338→    setProgress(function(p) { if (!p) return p; const next = markLearned(p, wordId); saveProgress(next); return next; });
   339→  }, []);
   340→  const recordQuizScore = useCallback(function(score) {
   341→    setProgress(function(p) { if (!p) return p; const next = score > p.quizHighScore ? Object.assign({}, p, { quizHighScore: score }) : p; saveProgress(next); return next; });
   342→  }, []);
   343→  const recordMatchingScore = useCallback(function(score) {
   344→    setProgress(function(p) { if (!p) return p; const next = score > p.matchingHighScore ? Object.assign({}, p, { matchingHighScore: score }) : p; saveProgress(next); return next; });
   345→  }, []);
   346→  const reset = useCallback(function() {
   347→    const fresh = defaultProgress(); saveProgress(fresh); setProgress(fresh);
   348→  }, []);
   349→  return { progress, hydrated, grade, markLearnedWord, recordQuizScore, recordMatchingScore, reset };
   350→}

   353→function shuffle(arr) {
   354→  const a = arr.slice();
   355→  for (let i = a.length - 1; i > 0; i--) {
   356→    const j = Math.floor(Math.random() * (i + 1));
   357→    const tmp = a[i]; a[i] = a[j]; a[j] = tmp;
   358→  }
   359→  return a;
   360→}

   362→function speak(text) {
   363→  if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   364→  const u = new SpeechSynthesisUtterance(text);
   365→  u.lang = "zh-CN"; u.rate = 0.9;
   366→  const voices = window.speechSynthesis.getVoices();
   367→  const zh = voices.find(function(v) { return v.lang.startsWith("zh"); });
   368→  if (zh) u.voice = zh;
   369→  window.speechSynthesis.cancel();
   370→  window.speechSynthesis.speak(u);
   371→}

   524→function Flashcard(props) {
   525→  const topicId = props.topicId || "all";
   526→  const onGrade = props.onGrade;
   527→  const onMarkLearned = props.onMarkLearned;
   528→  const dueWordIds = props.dueWordIds;
   529→  const [deck, setDeck] = useState([]);
   530→  const [index, setIndex] = useState(0);
   531→  const [flipped, setFlipped] = useState(false);
   532→
   533→  useEffect(function() {
   534→    let words = [];
   535→    if (topicId === "all") words = VOCAB.slice();
   536→    else if (topicId === "due") {
   537→      const ids = new Set(dueWordIds || []);
   538→      words = VOCAB.filter(function(w) { return ids.has(w.id); });
   539→    } else {
   540→      words = VOCAB.filter(function(w) { return w.topic === topicId; });
   541→    }
   542→    setDeck(words);
   543→    setIndex(0);
   544→    setFlipped(false);
   545→  }, [topicId, dueWordIds]);
   546→
   547→  const current = deck[index];
   548→
   549→  const goNext = useCallback(function() {
   550→    setFlipped(false);
   551→    setTimeout(function() { setIndex(function(i) { return (i + 1) % Math.max(deck.length, 1); }); }, 120);
   552→  }, [deck.length]);
   553→
   554→  const goPrev = useCallback(function() {
   555→    setFlipped(false);
   556→    setTimeout(function() { setIndex(function(i) { return (i - 1 + deck.length) % Math.max(deck.length, 1); }); }, 120);
   557→  }, [deck.length]);
   558→
   559→  const handleGrade = useCallback(function(grade) {
   560→    if (!current) return;
   561→    if (onGrade) onGrade(current.id, grade);
   562→    if (grade >= 3 && onMarkLearned) onMarkLearned(current.id);
   563→    goNext();
   564→  }, [current, onGrade, onMarkLearned, goNext]);
   565→
   566→  useEffect(function() {
   567→    function handler(e) {
   568→      if (!current) return;
   569→      if (e.code === "Space") { e.preventDefault(); setFlipped(function(f) { return !f; }); }
   570→      else if (e.code === "ArrowRight") goNext();
   571→      else if (e.code === "ArrowLeft") goPrev();
   572→      else if (e.code === "Digit1") handleGrade(0);
   573→      else if (e.code === "Digit2") handleGrade(3);
   574→      else if (e.code === "Digit3") handleGrade(5);
   575→    }
   576→    window.addEventListener("keydown", handler);
   577→    return function() { window.removeEventListener("keydown", handler); };
   578→  }, [current, goNext, goPrev, handleGrade]);
   579→
   580→  if (deck.length === 0 || !current) {
   581→    return html`
   582→      <div className="flex flex-col items-center justify-center py-20 text-center">
   583→        <div className="text-6xl mb-4">🎉</div>
   584→        <h3 className="text-xl font-semibold mb-2">Tuyệt vời!</h3>
   585→        <p className="text-gray-600">${topicId === "due" ? "Không có từ nào cần ôn lúc này. Hãy quay lại sau hoặc học chủ đề mới!" : "Chưa có từ vựng trong bộ này."}</p>
   586→      </div>`;
   587→  }
   588→
   589→  const topic = getTopic(current.topic);
   590→
   591→  return html`
   592→    <div className="flex flex-col gap-6">
   593→      <div className="space-y-2">
   594→        <div className="flex items-center justify-between text-sm text-gray-500">
   595→          <span>Thẻ ${index + 1} / ${deck.length}</span>
   596→          <span>${Math.round(((index + 1) / deck.length) * 100)}%</span>
   597→        </div>
   598→        <div className="h-2 rounded-full bg-gray-200 overflow-hidden">
   599→          <div className="h-full bg-gradient-to-r from-violet-500 to-fuchsia-500 transition-all" style=${{width: ((index + 1) / deck.length * 100) + "%"}}></div>
   600→        </div>
   601→      </div>
   602→
   603→      <div className=${"flip-card w-full h-[420px] cursor-pointer" + (flipped ? " flipped" : "")} onClick=${function() { setFlipped(function(f) { return !f; }); }}>
   604→        <div className="flip-card-inner">
   605→          <div className=${"flip-card-front rounded-3xl flex flex-col items-center justify-center gap-4 p-8 bg-gradient-to-br " + topic.color + " shadow-xl text-white overflow-hidden"}>
   606→            <div className="absolute top-4 left-4">
   607→              <span className="px-2 py-1 rounded-full bg-white/25 text-white text-xs">${topic.emoji} ${topic.name}</span>
   608→            </div>
   609→            <button className="absolute top-4 right-4 text-white hover:bg-white/20 p-2 rounded-full" onClick=${function(e) { e.stopPropagation(); speak(current.han); }}>🔊</button>
   610→            <div className="text-7xl mb-2">${current.emoji}</div>

   614→          </div>

   616→            <div className="flex items-center justify-between">
   617→              <span className="px-2 py-1 rounded-full bg-gray-100 text-xs">${current.pos}</span>

   652→const QUIZ_TIME = 15;

   655→function Quiz(props) {
   656→  const onQuizComplete = props.onQuizComplete;
   657→  const [questions, setQuestions] = useState([]);
   658→  const [current, setCurrent] = useState(0);
   659→  const [selected, setSelected] = useState(null);
   660→  const [score, setScore] = useState(0);
   661→  const [timeLeft, setTimeLeft] = useState(QUIZ_TIME);
   662→  const [phase, setPhase] = useState("intro");
   663→  const completedRef = useRef(false);
   664→
   665→  const generateQuiz = useCallback(function() {
   666→    const pool = shuffle(VOCAB).slice(0, QUIZ_COUNT);
   667→    const modes = ["han-to-vi", "vi-to-han", "pinyin-to-han"];
   668→    const qs = pool.map(function(word) {
   669→      const mode = modes[Math.floor(Math.random() * modes.length)];
   670→      const distractors = shuffle(VOCAB.filter(function(w) { return w.id !== word.id; })).slice(0, 3);
   671→      let correct, options;
   672→      if (mode === "han-to-vi") {
   673→        correct = word.meaning;
   674→        options = shuffle([correct].concat(distractors.map(function(d) { return d.meaning; })));
   675→      } else if (mode === "vi-to-han") {
   676→        correct = word.han;
   677→        options = shuffle([correct].concat(distractors.map(function(d) { return d.han; })));
   678→      } else {
   679→        correct = word.han;
   680→        options = shuffle([correct].concat(distractors.map(function(d) { return d.han; })));
   681→      }
   682→      return { word, options, correct, mode };
   683→    });
   684→    setQuestions(qs);
   685→    setCurrent(0); setSelected(null); setScore(0); setTimeLeft(QUIZ_TIME);
   686→    completedRef.current = false;
   687→  }, []);
   688→
   689→  function start() { generateQuiz(); setPhase("playing"); }
   690→
   691→  useEffect(function() {
   692→    if (phase !== "playing") return;
   693→    if (selected !== null) return;
   694→    if (timeLeft <= 0) { setSelected("__timeout__"); return; }
   695→    const t = setTimeout(function() { setTimeLeft(function(v) { return v - 1; }); }, 1000);
   696→    return function() { clearTimeout(t); };
   697→  }, [phase, selected, timeLeft]);
   698→
   699→  function handleSelect(opt) {
   700→    if (selected !== null) return;
   701→    setSelected(opt);
   702→    if (opt === questions[current].correct) setScore(function(s) { return s + 1; });
   703→  }
   704→
   705→  function next() {
   706→    if (current + 1 >= questions.length) {
   707→      setPhase("result");
   708→      return;
   709→    }
   710→    setCurrent(function(c) { return c + 1; });
   711→    setSelected(null);
   712→    setTimeLeft(QUIZ_TIME);
   713→  }
   714→
   715→  useEffect(function() {
   716→    if (phase === "result" && !completedRef.current && questions.length > 0) {
   717→      completedRef.current = true;
   718→      if (onQuizComplete) onQuizComplete(score, questions.length);
   719→    }
   720→    if (phase !== "result") completedRef.current = false;
   721→  }, [phase, score, questions.length, onQuizComplete]);
   722→
   723→  if (phase === "intro") {
   724→    return html`
   725→      <div className="p-8 flex flex-col items-center gap-6 text-center rounded-3xl bg-gradient-to-br from-violet-100 via-fuchsia-50 to-pink-100">
   726→        <div className="w-20 h-20 rounded-full bg-gradient-to-br from-violet-500 to-fuchsia-500 flex items-center justify-center shadow-lg text-4xl">🏆</div>
   727→        <div>
   728→          <h3 className="text-2xl font-bold mb-2">Quiz trắc nghiệm</h3>
   729→          <p className="text-gray-600 max-w-md">${QUIZ_COUNT} câu hỏi • ${QUIZ_TIME} giây/câu • 3 dạng: Hán → Việt, Việt → Hán, Pinyin → Hán</p>
   730→        </div>
   731→        <button onClick=${start} className="px-6 py-3 rounded-xl bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white hover:opacity-90 font-medium">Bắt đầu quiz 🔄</button>
   732→      </div>`;
   733→  }
   734→
   735→  if (phase === "result") {
   736→    const pct = Math.round((score / questions.length) * 100);
   737→    const emoji = pct >= 80 ? "🏆" : pct >= 60 ? "🎉" : pct >= 40 ? "💪" : "📚";
   738→    const title = pct >= 80 ? "Xuất sắc!" : pct >= 60 ? "Tốt lắm!" : pct >= 40 ? "Cố lên!" : "Cần luyện thêm!";
   739→    return html`
   740→      <div className="p-8 flex flex-col items-center gap-6 text-center rounded-3xl bg-gradient-to-br from-emerald-50 via-teal-50 to-cyan-50">
   741→        <div className="text-7xl">${emoji}</div>
   742→        <div>
   743→          <h3 className="text-3xl font-bold mb-1">${title}</h3>
   744→          <p className="text-gray-600">Bạn đã trả lời đúng</p>
   745→        </div>
   746→        <div className="text-6xl font-bold bg-gradient-to-r from-emerald-500 to-teal-500 bg-clip-text text-transparent">${score}/${questions.length}</div>
   747→        <span className="px-4 py-1 rounded-full bg-gray-100 text-base">${pct}% chính xác</span>
   748→        <div className="flex gap-3">
   749→          <button onClick=${start} className="px-4 py-2 rounded-xl bg-gradient-to-r from-emerald-500 to-teal-500 text-white">🔄 Làm lại</button>
   750→          <button onClick=${function() { setPhase("intro"); }} className="px-4 py-2 rounded-xl border border-gray-300">Về trang quiz</button>
   751→        </div>
   752→      </div>`;
   753→  }
   754→
   755→  const q = questions[current];
   756→  if (!q) return null;
   757→  const promptLabel = q.mode === "han-to-vi" ? "Chọn nghĩa tiếng Việt" : q.mode === "vi-to-han" ? "Chọn Hán tự đúng" : "Chọn Hán tự theo pinyin";
   758→
   759→  return html`
   760→    <div className="flex flex-col gap-5">
   761→      <div className="flex items-center justify-between gap-4">
   762→        <span className="px-2 py-1 rounded-full bg-gray-100 text-sm">Câu ${current + 1} / ${questions.length}</span>
   763→        <div className="flex items-center gap-2">
   764→          <span className="px-2 py-1 rounded-full bg-amber-100 text-amber-700 text-sm">⏱ ${timeLeft}s</span>
   765→          <span className="px-2 py-1 rounded-full bg-emerald-100 text-emerald-700 text-sm">🏆 ${score}</span>
   766→        </div>
   767→      </div>
   768→
   769→      <div className="h-1.5 rounded-full bg-amber-100 overflow-hidden">
   770→        <div className="h-full bg-amber-400 transition-all" style=${{width: (timeLeft / QUIZ_TIME * 100) + "%"}}></div>
   771→      </div>
   772→
   773→      <div className="p-6 md:p-8 flex flex-col items-center gap-4 rounded-3xl bg-gradient-to-br from-violet-50 to-fuchsia-50 border-2 border-violet-100">
   774→        <div className="text-xs uppercase tracking-wider text-gray-500">${promptLabel}</div>
   775→        <div className="flex items-center gap-3">
   776→          <div className="text-5xl md:text-6xl font-bold">
   777→            ${q.mode === "han-to-vi" ? q.word.han : null}
   778→            ${q.mode === "vi-to-han" ? q.word.meaning : null}
   779→            ${q.mode === "pinyin-to-han" ? html`<span className="italic">${q.word.pinyin}</span>` : null}
   780→          </div>
   781→          ${q.mode === "han-to-vi" ? html`<button className="p-2 hover:bg-gray-100 rounded-full" onClick=${function() { speak(q.word.han); }}>🔊</button>` : null}
   782→        </div>
   783→        ${q.mode === "han-to-vi" ? html`<div className="text-base italic text-gray-500">${q.word.pinyin}</div>` : null}
   784→        ${q.mode === "vi-to-han" ? html`<div className="text-sm text-gray-500">${q.word.emoji} ${q.word.pos}</div>` : null}
   785→      </div>
   786→
   787→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   788→        ${q.options.map(function(opt) {
   789→          const isCorrect = opt === q.correct;
   790→          const isSelected = opt === selected;
   791→          let cls = "border-2 hover:border-violet-300 hover:bg-violet-50 transition-all justify-start text-left";

   793→            if (isCorrect) cls = "border-2 border-emerald-400 bg-emerald-50 text-emerald-700 justify-start text-left";
   794→            else if (isSelected) cls = "border-2 border-rose-400 bg-rose-50 text-rose-700 justify-start text-left";
   795→            else cls = "border-2 border-gray-200 opacity-60 justify-start text-left";
   796→          }

   798→            <button key=${opt} className=${"py-4 px-4 text-lg rounded-xl " + cls} onClick=${function() { handleSelect(opt); }} disabled=${selected !== null}>
   799→              <span className="flex-1">${opt}</span>

   803→        })}

   806→      ${selected !== null ? html`
   807→        <div className="flex justify-end">
   808→          <button onClick=${next} className="px-4 py-2 rounded-xl bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white">
   809→            ${current + 1 >= questions.length ? "Xem kết quả" : "Câu tiếp theo"}
   810→          </button>
   811→        </div>
   812→      ` : null}
   813→    </div>`;

   822→  const [phase, setPhase] = useState("intro");
   823→  const [cells, setCells] = useState([]);
   824→  const [matched, setMatched] = useState(new Set());
   825→  const [selected, setSelected] = useState(null);
   826→  const [wrongPair, setWrongPair] = useState(null);
   827→  const [lives, setLives] = useState(MATCH_LIVES);
   828→  const [seconds, setSeconds] = useState(0);
   829→  const completedRef = useRef(false);
   830→
   831→  const start = useCallback(function() {
   832→    const pool = shuffle(VOCAB).slice(0, MATCH_PAIRS);
   833→    const hanCells = pool.map(function(w) { return { id: "h-" + w.id, wordId: w.id, side: "han", text: w.han, pinyin: w.pinyin, emoji: w.emoji }; });
   834→    const viCells = pool.map(function(w) { return { id: "v-" + w.id, wordId: w.id, side: "vi", text: w.meaning }; });
   835→    setCells(shuffle(hanCells.concat(viCells)));
   836→    setMatched(new Set()); setSelected(null); setWrongPair(null);
   837→    setLives(MATCH_LIVES); setSeconds(0); setPhase("playing");
   838→    completedRef.current = false;
   839→  }, []);
   840→
   841→  useEffect(function() {
   842→    if (phase !== "playing") return;
   843→    const t = setInterval(function() { setSeconds(function(s) { return s + 1; }); }, 1000);
   844→    return function() { clearInterval(t); };
   845→  }, [phase]);
   846→
   847→  useEffect(function() {
   848→    if (phase !== "playing") return;
   849→    if (cells.length > 0 && matched.size === cells.length) setPhase("won");
   850→  }, [matched, cells.length, phase]);
   851→
   852→  useEffect(function() {
   853→    if (phase !== "playing") return;
   854→    if (lives <= 0) setPhase("lost");
   855→  }, [lives, phase]);
   856→
   857→  useEffect(function() {
   858→    if (phase === "won" && !completedRef.current) {
   859→      completedRef.current = true;
   860→      const score = Math.max(0, 1000 - seconds * 5 + lives * 100);
   861→      if (onComplete) onComplete(score);
   862→    }
   863→    if (phase !== "won") completedRef.current = false;
   864→  }, [phase, seconds, lives, onComplete]);
   865→
   866→  function handleClick(cell) {
   867→    if (matched.has(cell.id) || wrongPair) return;
   868→    if (!selected) { setSelected(cell); return; }
   869→    if (selected.id === cell.id) { setSelected(null); return; }
   870→    if (selected.side === cell.side) { setSelected(cell); return; }
   871→    if (selected.wordId === cell.wordId) {
   872→      const nm = new Set(matched); nm.add(selected.id); nm.add(cell.id);
   873→      setMatched(nm); setSelected(null);
   874→    } else {
   875→      setWrongPair([selected.id, cell.id]);
   876→      setLives(function(l) { return l - 1; });
   877→      setTimeout(function() { setWrongPair(null); setSelected(null); }, 800);
   878→    }
   879→  }
   880→
   881→  if (phase === "intro") {
   882→    return html`
   883→      <div className="p-8 flex flex-col items-center gap-6 text-center rounded-3xl bg-gradient-to-br from-teal-100 via-cyan-50 to-sky-100">
   884→        <div className="w-20 h-20 rounded-full bg-gradient-to-br from-teal-500 to-cyan-500 flex items-center justify-center shadow-lg text-4xl">✓</div>
   885→        <div>
   886→          <h3 className="text-2xl font-bold mb-2">Ghép cặp từ</h3>
   887→          <p className="text-gray-600 max-w-md">Ghép ${MATCH_PAIRS} cặp Hán tự ↔ Nghĩa tiếng Việt. ${MATCH_LIVES} mạng, càng nhanh điểm càng cao!</p>
   888→        </div>
   889→        <button onClick=${start} className="px-6 py-3 rounded-xl bg-gradient-to-r from-teal-500 to-cyan-500 text-white hover:opacity-90 font-medium">Bắt đầu ghép cặp 🔄</button>
   890→      </div>`;
   891→  }
   892→
   893→  if (phase === "won" || phase === "lost") {
   894→    const score = phase === "won" ? Math.max(0, 1000 - seconds * 5 + lives * 100) : 0;
   895→    return html`
   896→      <div className=${"p-8 flex flex-col items-center gap-6 text-center rounded-3xl " + (phase === "won" ? "bg-gradient-to-br from-emerald-100 via-teal-50 to-cyan-100" : "bg-gradient-to-br from-rose-100 via-pink-50 to-red-100")}>
   897→        <div className="text-7xl">${phase === "won" ? "🏆" : "💔"}</div>
   898→        <div>
   899→          <h3 className="text-3xl font-bold mb-1">${phase === "won" ? "Chiến thắng!" : "Hết mạng!"}</h3>
   900→          <p className="text-gray-600">${phase === "won" ? "Hoàn thành trong " + seconds + "s, còn " + lives + " mạng" : "Đừng nản, thử lại nhé!"}</p>
   901→        </div>
   902→        ${phase === "won" ? html`<div className="text-5xl font-bold bg-gradient-to-r from-emerald-500 to-teal-500 bg-clip-text text-transparent">${score} điểm</div>` : null}
   903→        <div className="flex gap-3">
   904→          <button onClick=${start} className="px-4 py-2 rounded-xl bg-gradient-to-r from-teal-500 to-cyan-500 text-white">🔄 Chơi lại</button>
   905→          <button onClick=${function() { setPhase("intro"); }} className="px-4 py-2 rounded-xl border border-gray-300">Về trang ghép cặp</button>
   906→        </div>
   907→      </div>`;
   908→  }
   909→
   910→  return html`
   911→    <div className="flex flex-col gap-5">
   912→      <div className="flex items-center justify-between gap-4">
   913→        <div className="flex items-center gap-2">
   914→          ${Array.from({ length: MATCH_LIVES }).map(function(_, i) {
   915→            return html`<span key=${i} className=${"text-xl " + (i < lives ? "" : "opacity-30 grayscale")}>❤️</span>`;
   916→          })}
   917→        </div>
   918→        <div className="flex items-center gap-2">
   919→          <span className="px-2 py-1 rounded-full bg-sky-100 text-sky-700 text-sm">⏱ ${seconds}s</span>
   920→          <span className="px-2 py-1 rounded-full bg-emerald-100 text-emerald-700 text-sm">✓ ${matched.size / 2}/${MATCH_PAIRS}</span>
   921→        </div>
   922→      </div>
   923→
   924→      <div className="grid grid-cols-2 gap-3 md:gap-4">
   925→        ${cells.map(function(cell) {
   926→          const isMatched = matched.has(cell.id);
   927→          const isSelected = selected && selected.id === cell.id;
   928→          const isWrong = wrongPair && wrongPair.indexOf(cell.id) !== -1;
   929→          let cls = "bg-white border-slate-200 hover:border-violet-300 hover:shadow-md";
   930→          if (isMatched) cls = "bg-emerald-100 border-emerald-300 opacity-50";
   931→          else if (isWrong) cls = "bg-rose-100 border-rose-400 pulse-wrong";
   932→          else if (isSelected) cls = "bg-gradient-to-br from-violet-500 to-fuchsia-500 border-violet-600 text-white shadow-lg scale-105";
   933→          return html`
   934→            <button key=${cell.id} onClick=${function() { handleClick(cell); }} disabled=${isMatched} className=${"relative rounded-2xl p-4 md:p-5 min-h-[100px] md:min-h-[120px] flex flex-col items-center justify-center gap-1 transition-all border-2 " + cls}>
   935→              ${cell.side === "han" ? html`
   936→                <div>
   937→                  <div className="text-3xl">${cell.emoji}</div>
   938→                  <div className="text-2xl md:text-3xl font-bold">${cell.text}</div>
   939→                  ${!isMatched ? html`<div className=${"text-xs italic " + (isSelected ? "text-white/80" : "text-gray-500")}>${cell.pinyin}</div>` : null}
   940→                </div>
   941→              ` : html`<div className="text-base md:text-lg font-semibold text-center">${cell.text}</div>`}
   942→              ${isMatched ? html`<span className="absolute top-2 right-2 text-emerald-600">✓</span>` : null}
   943→            </button>`;
   944→        })}
   945→      </div>
   946→
   947→      <div className="text-center text-xs text-gray-500">Nhấp vào 1 từ Hán và 1 nghĩa tiếng Việt để ghép cặp</div>
   948→    </div>`;
   949→}
